## 问题1: 结构体指针 vs 指向结构体成员的指针

### 面试问题
- 结构体指针和指向结构体成员的指针有什么区别？
- 在BootLoader项目中，如何使用结构体指针高效解析mkimage头部信息？

### 回答关键点

#### 1. 技术概念对比
```c
// 结构体指针方式（推荐）

// 指向成员的指针方式（不推荐）
```

## 问题2: volatile关键字的应用

### 面试问题
- 为什么硬件寄存器访问必须使用volatile？
- 在多任务环境下，仅使用volatile是否足够保证变量的安全访问？

### 回答关键点

#### 1. 硬件寄存器访问
```c
// 错误示例：编译器可能缓存寄存器值
uint32_t *gpio_odr = (uint32_t *)0x40020014;

// 正确示例：强制每次从寄存器读取
volatile uint32_t *gpio_odr = (volatile uint32_t *)0x40020014;
```

**可能导致的问题:**
- 寄存器状态不同步
- 中断标志清除失败
- 硬件自动清除的状态位读取错误

#### 2. 多任务环境局限性
```c
volatile int g_sharedData = 0;
g_sharedData++;  // 非原子操作！读-改-写可能被任务切换中断
```

**解决方案:**
- **关中断保护**: 适合简单快速操作
- **互斥锁**: 适合复杂临界区，支持优先级继承
- **信号量**: 适合任务同步
- **原子操作**: 硬件支持的原子操作

#### 3. 面试表达要点
- "volatile只解决可见性问题，不解决原子性问题"
- "在我的项目中遇到过这样的情况..."
- "选择同步机制时需要考虑性能和安全性的平衡"

## 问题3: CRC 校验与固件安全 (BootLoader 核心)

### 面试问题
- 简述CRC校验的核心思想和工作原理
- 在BootLoader中，除了CRC校验，还有哪些机制可以验证固件的来源可靠性？

### 回答关键点

#### 1. CRC校验原理
```c
// CRC核心思想：多项式除法
// 发送端：数据 + 生成多项式 → 余数(CRC码)
// 接收端：(数据+CRC码) ÷ 生成多项式 → 余数为0则正确

// 项目中的应用
head = (image_header_t *)pos;
uint32_t stored_crc = be32_to_cpu(head->ih_dcrc);  // 存储的CRC
uint32_t calculated_crc = crc32(data, size);       // 计算的CRC
if (stored_crc != calculated_crc) {
    printf("CRC check failed!\r\n");
    return -1;
}
```

**CRC特点:**
- **检错能力强**: 能检测单比特错误、突发错误
- **计算简单**: 硬件实现效率高
- **标准化**: CRC32、CRC16等广泛应用

## 问题4: HMAC原理与实现 (消息认证码)

### 面试问题
- 什么是HMAC？它是如何工作的？
- 相比简单哈希和数字签名，HMAC有什么优势和局限性？
- 在BootLoader中如何使用HMAC验证固件来源？

### 回答关键点

#### 1. HMAC基本原理
```c
// HMAC核心思想：使用共享密钥和哈希函数生成认证码
// HMAC(K, M) = H((K ⊕ opad) || H((K ⊕ ipad) || M))
// 其中：K=密钥, M=消息, H=哈希函数, ipad=0x36, opad=0x5C

// HMAC计算步骤
void hmac_sha256(const uint8_t *key, size_t key_len,
                 const uint8_t *data, size_t data_len,
                 uint8_t *hmac_out)
{
    uint8_t k_ipad[64], k_opad[64];  // SHA-256块大小64字节
    uint8_t inner_hash[32];          // SHA-256输出32字节
    
    // 步骤1: 处理密钥长度并创建填充密钥
    process_key(key, key_len, k_ipad, k_opad);
    
    // 步骤2: 计算内部哈希 H((K ⊕ ipad) || M)
    sha256_context ctx;
    sha256_init(&ctx);
    sha256_update(&ctx, k_ipad, 64);
    sha256_update(&ctx, data, data_len);
    sha256_final(&ctx, inner_hash);
    
    // 步骤3: 计算最终HMAC H((K ⊕ opad) || inner_hash)
    sha256_init(&ctx);
    sha256_update(&ctx, k_opad, 64);
    sha256_update(&ctx, inner_hash, 32);
    sha256_final(&ctx, hmac_out);
}
```

**HMAC提供的安全保证:**
- **完整性验证**: 检测数据是否被篡改
- **身份认证**: 验证消息来自拥有密钥的发送方
- **防重放攻击**: 结合时间戳或序列号使用

#### 2. 在BootLoader中的应用设计

**扩展固件头部结构:**
```c
typedef struct secure_image_header {
    image_header_t basic_header;    // 原有头部结构
    uint8_t hmac[32];              // HMAC-SHA256值
    uint32_t timestamp;            // 时间戳（防重放）
    uint32_t sequence;             // 序列号
    uint8_t reserved[24];          // 预留字段，保持对齐
} secure_image_header_t;

// 预置在BootLoader中的共享密钥
static const uint8_t FIRMWARE_HMAC_KEY[32] = {
    0x12, 0x34, 0x56, 0x78, 0x9A, 0xBC, 0xDE, 0xF0,
    // ... 其余24字节
};
```

**固件验证流程:**
```c
int verify_firmware_hmac(unsigned int pos)
{
    secure_image_header_t *sec_head = (secure_image_header_t *)pos;
    
    // 1. 基本魔数验证
    if (be32_to_cpu(sec_head->basic_header.ih_magic) != IH_MAGIC) {
        printf("魔数验证失败\r\n");
        return -1;
    }
    
    // 2. 获取固件数据
    uint32_t size = be32_to_cpu(sec_head->basic_header.ih_size);
    uint8_t *firmware_data = (uint8_t *)(pos + sizeof(secure_image_header_t));
    
    // 3. 构造待验证的消息（头部基本信息 + 固件数据）
    uint8_t calculated_hmac[32];
    hmac_sha256_context ctx;
    hmac_sha256_init(&ctx, FIRMWARE_HMAC_KEY, 32);
    
    // 包含基本头部信息（除了HMAC字段）
    hmac_sha256_update(&ctx, (uint8_t*)&sec_head->basic_header, sizeof(image_header_t));
    hmac_sha256_update(&ctx, (uint8_t*)&sec_head->timestamp, 4);
    hmac_sha256_update(&ctx, (uint8_t*)&sec_head->sequence, 4);
    
    // 包含固件数据
    hmac_sha256_update(&ctx, firmware_data, size);
    hmac_sha256_final(&ctx, calculated_hmac);
    
    // 4. 安全比较HMAC值（防时序攻击）
    if (!constant_time_compare(sec_head->hmac, calculated_hmac, 32)) {
        printf("HMAC验证失败！固件可能来自非法来源\r\n");
        return -1;
    }
    
    printf("HMAC验证成功，固件来源可信\r\n");
    return 0;
}

// 防时序攻击的常量时间比较
int constant_time_compare(const uint8_t *a, const uint8_t *b, size_t len)
{
    volatile uint8_t result = 0;
    for (size_t i = 0; i < len; i++) {
        result |= a[i] ^ b[i];  // 总是比较所有字节
    }
    return result == 0;
}
```

### 回答关键点

#### 1. 字节序基本概念

**大端序 vs 小端序:**
```c
// 以32位数值0x12345678为例
uint32_t value = 0x12345678;

// 大端序(Big-Endian) - 网络字节序
// 高位字节存储在低地址
// 内存布局: [0x12][0x34][0x56][0x78]
//          地址0  地址1  地址2  地址3

// 小端序(Little-Endian) - 主机字节序(x86, ARM等)
// 低位字节存储在低地址  
// 内存布局: [0x78][0x56][0x34][0x12]
//          地址0  地址1  地址2  地址3
```

**判断系统字节序:**
```c
// 运行时检测字节序
int is_little_endian(void)
{
    uint32_t test = 0x12345678;
    uint8_t *p = (uint8_t *)&test;
    
    if (p[0] == 0x78) {
        return 1;  // 小端序
    } else {
        return 0;  // 大端序
    }
}

// 编译时检测(GCC)
#if __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
    #define SYSTEM_LITTLE_ENDIAN 1
#else
    #define SYSTEM_LITTLE_ENDIAN 0
#endif
```

#### 3. 内存对齐重要性

**对齐问题的影响:**
```c
// 未对齐访问可能导致的问题
typedef struct {
    uint8_t  flag;      // 1字节
    uint32_t address;   // 4字节，可能未对齐
    uint16_t length;    // 2字节
} unaligned_struct_t;   // 总大小可能是7字节

// 在某些ARM Cortex-M0上，未对齐访问会产生硬件异常
void alignment_fault_example(void)
{
    uint8_t buffer[10];
    uint32_t *p = (uint32_t *)(buffer + 1);  // 未对齐地址
```

```c
typedef struct {
    uint8_t  flag;
    uint32_t address   __attribute__((aligned(4)));  // 强制4字节对齐
    uint16_t length;
} __attribute__((packed)) aligned_struct_t;

// 或使用#pragma pack
#pragma pack(1)  // 1字节对齐，紧密打包
typedef struct {
    uint8_t  flag;
    uint32_t address;
    uint16_t length;
} packed_struct_t;
#pragma pack()

// 安全的未对齐访问
uint32_t safe_read_unaligned(uint8_t *addr)
{
    return (addr[0] << 0)  |
           (addr[1] << 8)  |
           (addr[2] << 16) |
           (addr[3] << 24);
}

void safe_write_unaligned(uint8_t *addr, uint32_t value)
{
    addr[0] = (value >> 0)  & 0xFF;
    addr[1] = (value >> 8)  & 0xFF;
    addr[2] = (value >> 16) & 0xFF;
    addr[3] = (value >> 24) & 0xFF;
}
```

## 问题7: 中断服务程序 (ISR) 设计原则 (硬件交互关键)

### 面试问题
- 嵌入式开发中，设计中断服务程序 (ISR) 需要遵循"快进快出"原则。请解释为什么这条原则如此重要？
- 在 ISR 中执行耗时操作会导致什么问题？请结合 STM32 的中断优先级和嵌套机制说明
- 在你的项目中，当 ISR 需要将数据传递给任务处理时，使用哪些 FreeRTOS 机制？为什么不能直接使用普通的队列发送函数？

### 回答关键点

#### 1. "快进快出"原则的重要性

**核心概念:**
```c
// 错误的ISR设计（违反快进快出原则）
void USART1_IRQHandler(void)
{
    uint8_t data = USART1->DR;
    
    // 错误：在ISR中执行复杂处理
    process_protocol_data(data);        // 可能耗时数毫秒
    calculate_checksum(buffer, 1024);   // 大量计算
    HAL_Delay(10);                      // 阻塞延时！
    printf("Received: %c\r\n", data);   // I/O操作
}

// 正确的ISR设计
void USART1_IRQHandler(void)
{
    uint8_t data = USART1->DR;
    
    // 快速处理：只做必要的硬件操作
    if (rx_head < RX_BUFFER_SIZE) {
        rx_buffer[rx_head++] = data;
        data_received_flag = 1;  // 设置标志位
    }
    
    // 清除中断标志
    USART1->SR &= ~USART_SR_RXNE;
    
    // 可选：唤醒等待任务
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xSemaphoreGiveFromISR(uart_semaphore, &xHigherPriorityTaskWoken);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}
```

**快进快出原则的重要性:**
- **响应性保证**: 确保系统能及时响应其他中断
- **实时性维护**: 防止高优先级中断被长时间阻塞
- **系统稳定性**: 避免中断嵌套过深导致栈溢出
- **时序要求**: 硬件外设往往有严格的时序要求

#### 2. 耗时操作在ISR中的危害

**STM32中断优先级配置:**
```c
void NVIC_Configuration(void)
{
    // 配置中断优先级组
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_4);
    
    // 高优先级中断：系统定时器
    NVIC_InitStructure.NVIC_IRQChannel = SysTick_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;  // 最高优先级
    NVIC_Init(&NVIC_InitStructure);
    
    // 中等优先级：UART接收
    NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 2;
    NVIC_Init(&NVIC_InitStructure);
    
    // 低优先级：定时器
    NVIC_InitStructure.NVIC_IRQChannel = TIM2_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 3;
    NVIC_Init(&NVIC_InitStructure);
}
```

**具体危害分析:**
- **中断延迟累积**: 低优先级中断被长时间阻塞
- **栈溢出风险**: ISR中大数组消耗栈空间，中断嵌套加剧风险
- **实时性破坏**: 电机控制等实时任务被延迟，导致系统异常
- **系统响应性下降**: 用户交互响应变慢

#### 3. STM32中断优先级和嵌套机制

**抢占优先级机制:**
```c
// 中断嵌套示例
void TIM2_IRQHandler(void)
{
    // 开始执行TIM2 ISR（抢占优先级2）
    timer_processing_start();
    
    // 此时USART1中断到来（抢占优先级1）
    // TIM2 ISR被挂起，CPU转去执行USART1 ISR
    
    timer_processing_end();  // USART1 ISR执行完后才会执行到这里
}

void USART1_IRQHandler(void)
{
    // 这个ISR会抢占TIM2 ISR
    uint8_t data = USART1->DR;
    rx_buffer[rx_index++] = data;
    // 执行完毕后，CPU返回TIM2 ISR继续执行
}
```

**关键概念:**
- **抢占优先级**: 决定能否嵌套（数值越小优先级越高）
- **子优先级**: 同抢占优先级内的排序
- **中断嵌套**: 高优先级中断可以打断低优先级中断

#### 4. FreeRTOS中的ISR与任务通信

**ISR安全函数的必要性:**
```c
// 错误：在ISR中使用普通函数
void USART1_IRQHandler(void)
{
    uint8_t data = USART1->DR;
    
    // 错误！可能导致死锁
    xQueueSend(uart_queue, &data, 0);        // 不安全
    xSemaphoreGive(uart_semaphore);          // 不安全
    vTaskDelay(1);                           // 绝对禁止！
}

// 正确：使用ISR安全函数
void USART1_IRQHandler(void)
{
    uint8_t data = USART1->DR;
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    // 正确的ISR安全函数
    xQueueSendFromISR(uart_queue, &data, &xHigherPriorityTaskWoken);
    xSemaphoreGiveFromISR(uart_semaphore, &xHigherPriorityTaskWoken);
    
    // 如果有更高优先级任务被唤醒，进行任务切换
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}
```

**ISR安全函数的特殊处理:**
- **禁用调度器**: 不是禁用中断，而是禁用任务调度
- **快速检查**: 立即返回，不阻塞等待
- **延迟切换**: 标记需要任务切换，在ISR退出时进行
- **原子操作**: 直接操作内核数据结构，保证一致性

### 回答关键点

#### 1. UART关键配置参数

**必须配置的核心参数:**
```c
// UART初始化结构体示例
UART_HandleTypeDef huart1;

void UART_Init(void)
{
    huart1.Instance = USART1;
    
    // 1. 波特率 - 通信速度，必须双方一致
    huart1.Init.BaudRate = 115200;
    
    // 2. 数据位数 - 每个字符的数据位数
    huart1.Init.WordLength = UART_WORDLENGTH_8B;  // 8位数据
    
    // 3. 停止位 - 标记数据结束
    huart1.Init.StopBits = UART_STOPBITS_1;       // 1个停止位
    
    // 4. 校验位 - 错误检测机制
    huart1.Init.Parity = UART_PARITY_NONE;        // 无校验
    
    // 5. 硬件流控 - 防止数据溢出
    huart1.Init.HwFlowCtl = UART_HWCONTROL_NONE;   // 无硬件流控
    
    // 6. 工作模式 - 收发方向
    huart1.Init.Mode = UART_MODE_TX_RX;           // 收发模式
    
    // 7. 过采样率 - 提高接收可靠性
    huart1.Init.OverSampling = UART_OVERSAMPLING_16;  // 16倍过采样
    
    HAL_UART_Init(&huart1);
}
```

**参数选择的影响:**
- **波特率**: 影响传输速度和可靠性，常用9600/115200/460800
- **数据位**: 8位最常用，7位用于ASCII，9位用于地址标记
- **校验位**: EVEN/ODD校验能检测奇数个位错误
- **停止位**: 2个停止位提高可靠性但降低效率
- **流控**: RTS/CTS防止快速发送导致的数据丢失

```c
// 解决方案：内存池管理
typedef struct memory_pool {
    uint8_t *pool_start;
    uint32_t block_size;
    uint32_t block_count;
    uint8_t *free_list;
} memory_pool_t;

memory_pool_t* create_memory_pool(uint32_t block_size, uint32_t block_count)
{
    memory_pool_t *pool = pvPortMalloc(sizeof(memory_pool_t));
    if (pool == NULL) return NULL;
    
    pool->pool_start = pvPortMalloc(block_size * block_count);
    if (pool->pool_start == NULL) {
        vPortFree(pool);
        return NULL;
    }
    
    // 初始化固定大小块的空闲链表
    uint8_t **free_ptr = (uint8_t**)pool->pool_start;
    for (uint32_t i = 0; i < block_count - 1; i++) {
        free_ptr = (uint8_t**)(pool->pool_start + i * block_size);
        *free_ptr = pool->pool_start + (i + 1) * block_size;
    }
    free_ptr = (uint8_t**)(pool->pool_start + (block_count - 1) * block_size);
    *free_ptr = NULL;
    
    pool->free_list = pool->pool_start;
    return pool;
}
```

- STM32F4系列支持多种低功耗模式，在LoRa监测节点等IoT设备中，如何选择合适的低功耗模式？
- 在FreeRTOS系统中，如何协调任务调度与MCU的低功耗模式？进入Stop模式前需要保存哪些关键状态？
- 如何计算和优化IoT设备的功耗，以达到长期电池供电的要求？

### 回答关键点

#### 1. STM32F4低功耗模式选择策略

**不同模式的应用场景:**

// STM32F4低功耗模式详细对比
typedef enum {
    POWER_MODE_RUN,      // 正常运行: ~36mA @ 168MHz, 无延迟
    POWER_MODE_SLEEP,    // 睡眠模式: ~2.1mA, 唤醒<1μs, 外设继续工作
    POWER_MODE_STOP,     // 停止模式: ~112μA, 唤醒~5ms, 保持RAM和寄存器
    POWER_MODE_STANDBY   // 待机模式: ~2μA, 系统复位唤醒, 丢失RAM
} power_mode_t;

// 1. RUN模式 - 正常运行模式
void run_mode_characteristics(void)
{
    // 特点：
    // - CPU全速运行 (最高168MHz)
    // - 所有外设都可以工作
    // - 功耗最高: ~36mA @ 168MHz
    // - 无进入/退出延迟
    
    // 功耗构成：
    // - 核心功耗: ~25mA (CPU + 内存)
    // - GPIO功耗: ~1-2mA  
    // - 外设功耗: ~8-10mA (取决于使用的外设)
    
    // 适用场景：正常工作状态，需要高性能计算
}

// 2. SLEEP模式 - 睡眠模式
void sleep_mode_characteristics(void)
{
    // 特点：
    // - CPU停止，外设继续工作
    // - 所有RAM和寄存器内容保持
    // - 任何中断都可以唤醒
    // - 功耗: ~2.1mA
    // - 唤醒时间: <1μs (几个时钟周期)
    
    // 停止的部分：
    // - CPU内核停止执行
    // - 系统时钟(HCLK)停止
    
    // 保持的部分：
    // - 外设时钟继续运行
    // - 所有GPIO状态保持
    // - 所有SRAM内容保持
    // - 所有外设寄存器保持
    
    // 唤醒源：几乎所有中断
    // - 外部中断(EXTI) - UART/SPI/I2C中断
    // - 定时器中断 - ADC转换完成中断
    // - RTC中断 - 看门狗中断
    
    // 适用场景：短暂等待，需要快速响应
    HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);
}

// SLEEP模式的两种进入方式：WFI vs WFE
void sleep_entry_methods_comparison(void)
{
    // === 方式1: WFI (Wait For Interrupt) ===
    // 特点：只能被中断唤醒
    HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);
    
    // WFI唤醒条件：
    // - 任何使能的中断 (即使全局中断被禁用)
    // - NMI (不可屏蔽中断)
    // - 调试事件 (调试器连接时)
    
    // WFI关键特性：
    // - 即使__disable_irq()禁用全局中断，仍可被中断唤醒
    // - 唤醒后挂起的中断会在__enable_irq()后执行
    // - 唤醒机制简单明确，适合明确只需中断唤醒的场景
    
    // === 方式2: WFE (Wait For Event) ===
    // 特点：可被中断或事件唤醒
    HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFE);
    
    // WFE唤醒条件：
    // - 所有能唤醒WFI的中断
    // - 事件信号 (通过SEV指令)
    // - 外设事件 (DMA传输完成、定时器事件等)
    
    // WFE关键特性：
    // - 基于ARM Cortex-M的事件寄存器机制
    // - 支持多任务协作和外设事件唤醒
    // - 唤醒机制更灵活，适合需要多种唤醒源的场景
    
    // === 性能对比 ===
    // - 功耗：完全相同
    // - 唤醒延迟：基本相同 (<1μs)
    // - 代码复杂度：WFI更简单，WFE更灵活
    // - 调试友好性：WFI更容易调试，唤醒原因明确
}

// 实际应用中的选择策略
void wfi_wfe_selection_strategy(void)
{
    // 场景1: 等待特定中断 - 推荐WFI
    void wait_for_uart_interrupt(void)
    {
        printf("等待UART接收中断...\r\n");
        // 明确只需中断唤醒，使用WFI逻辑更清晰
        HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);
        printf("UART中断到达\r\n");
    }
    
    // 场景2: 等待DMA完成 - 推荐WFE  
    void wait_for_dma_completion(void)
    {
        HAL_SPI_Transmit_DMA(&hspi1, tx_buffer, size);
        printf("等待DMA传输完成...\r\n");
        // DMA完成产生事件信号，不一定产生中断
        HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFE);
        printf("DMA传输完成\r\n");
    }
    
    // 场景3: 多任务协作 - 推荐WFE
    void task_cooperation_example(void)
    {
        // 任务A完成工作后发送事件信号
        void taskA(void) {
            complete_work();
            __SEV();  // 发送事件信号通知其他任务
        }
        
        // 任务B等待任务A完成
        void taskB(void) {
            HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFE);
            continue_work();  // 被任务A的SEV信号唤醒
        }
    }
    
    // 场景4: 全局中断禁用时仍需唤醒 - 只能用WFI
    void interrupt_disabled_wakeup(void)
    {
        __disable_irq();  // 禁用全局中断
        NVIC_EnableIRQ(EXTI0_IRQn);  // 仍可配置特定中断
        
        // WFI仍可被中断唤醒，但中断服务程序不会执行
        HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);
        
        __enable_irq();   // 重新使能全局中断，处理挂起的中断
    }
}

// ARM Cortex-M事件寄存器机制详解
void event_register_mechanism(void)
{
    // ARM Cortex-M内部有一个1位的事件寄存器
    
    // 设置事件寄存器的方式：
    // 1. SEV指令 - 软件设置事件标志
    // 2. 外设事件 - 硬件设置 (DMA、定时器等)
    // 3. 中断信号 - 任何中断都会设置事件寄存器
    
    // WFE的执行逻辑：
    if (event_register_is_set) {
        // 如果事件寄存器已设置，WFE立即返回并清除寄存器
        clear_event_register();
        continue_execution();
    } else {
        // 如果事件寄存器未设置，进入睡眠等待
        enter_sleep_mode();
    }
    
    // 这解释了为什么有时需要连续两次WFE：
    __WFE();  // 第一次：清除可能存在的事件标志
    __WFE();  // 第二次：真正进入睡眠等待
    
    // 在实际STM32 HAL库中，这个细节已经被处理
    // HAL_PWR_EnterSLEEPMode会正确处理事件寄存器状态
}

// 3. STOP模式 - 停止模式  
void stop_mode_characteristics(void)
{
    // 特点：
    // - CPU和大部分外设停止
    // - RAM内容保持，寄存器内容保持
    // - 只有特定中断源可以唤醒
    // - 功耗: ~112μA (主调压器) 或 ~20μA (低功耗调压器)
    // - 唤醒时间: ~5ms (重新启动时钟)
    
    // 停止的部分：
    // - CPU内核完全停止
    // - 高速时钟(HSE, HSI, PLL)停止
    // - 系统时钟和外设时钟停止
    // - Flash进入掉电模式
    
    // 保持的部分：
    // - 所有SRAM内容 (192KB)
    // - 所有寄存器内容
    // - GPIO状态
    // - 备份域寄存器
    // - RTC继续工作(如果使用LSE/LSI)
    
    // 有限的唤醒源：
    // - EXTI线中断 (外部中断)
    // - RTC闹钟/唤醒定时器中断
    // - USB唤醒中断
    // - I2C唤醒中断 (从设备模式)
    
    // 适用场景：长时间等待，定时唤醒
    HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
}

// 4. STANDBY模式 - 待机模式
void standby_mode_characteristics(void)
{
    // 特点：
    // - 最低功耗模式
    // - 除备份域外，所有内容丢失
    // - 唤醒相当于系统复位
    // - 功耗: ~2μA
    // - 唤醒时间: 几毫秒 (重新初始化系统)
    
    // 关闭的部分：
    // - CPU内核完全关闭
    // - 所有时钟停止 (除LSE/LSI)
    // - 主调压器关闭
    // - 所有SRAM内容丢失 (192KB)
    // - 大部分寄存器复位
    // - GPIO状态丢失
    
    // 保持的部分：
    // - 备份域寄存器 (RTC_BKPxR)
    // - RTC计数器和配置
    // - LSE振荡器 (如果启用)
    // - 唤醒引脚检测电路
    
    // 极少的唤醒源：
    // - WKUP引脚上升沿 (PA0)
    // - RTC闹钟事件
    // - RTC唤醒定时器事件
    // - 外部复位 (NRST引脚)
    
    // 适用场景：长期休眠，最低功耗要求
    // 注意：唤醒后需要检查复位原因并恢复状态
    HAL_PWR_EnterSTANDBYMode();
}

        return POWER_MODE_SLEEP;    // 需要快速响应
    } else if (idle_time_ms < 100) {

    } else if (idle_time_ms < 60000) {
        return POWER_MODE_STOP;     // 中等空闲时间
    } else {
        return POWER_MODE_STANDBY;  // 长期空闲
    }
}

// LoRa监测节点的功耗场景分析
void lora_power_management_strategy(void)
{
    // 实际项目中的功耗模式应用
    
    // 场景1: 等待LoRa消息接收 (使用Sleep模式)
    // 原因: 需要快速响应LoRa DIO中断，外设(SPI)需要继续工作
    while (waiting_for_lora_packet) {
        HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);
        if (lora_packet_received) {
            process_lora_packet();  // 立即处理，无延迟
        }
    }
    
    // 场景2: 定时数据采集间隔 (使用Stop模式)  
    // 原因: 长时间等待(分钟级)，RTC可以在LSE时钟下继续工作
    setup_rtc_wakeup(300);  // 5分钟后唤醒
    save_peripheral_state(); // 保存SPI、UART等配置
    HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
    restore_peripheral_state(); // 恢复外设配置，重新启动时钟
    
    // 场景3: 设备维护或长期休眠 (使用Standby模式)
    // 原因: 最低功耗，通过WKUP引脚或RTC定时唤醒
    save_critical_data_to_backup_registers(); // 保存设备状态到备份寄存器
    setup_rtc_wakeup(24 * 3600); // 24小时后唤醒
    HAL_PWR_EnableWakeUpPin(PWR_WAKEUP_PIN1); // 使能WKUP引脚唤醒
    HAL_PWR_EnterSTANDBYMode(); // 进入待机模式，唤醒后系统复位
}

// STANDBY模式的唤醒处理
void handle_standby_wakeup(void)
{
    // 检查是否从STANDBY模式唤醒
    if (__HAL_PWR_GET_FLAG(PWR_FLAG_SB)) {
        printf("从STANDBY模式唤醒\r\n");
        __HAL_PWR_CLEAR_FLAG(PWR_FLAG_SB);
        
        // 读取备份寄存器恢复关键状态
        uint32_t magic_number = HAL_RTCEx_BKUPRead(&hrtc, RTC_BKP_DR0);
        if (magic_number == 0xDEADBEEF) {
            // 有效的备份数据，恢复设备状态
            uint32_t sensor_config = HAL_RTCEx_BKUPRead(&hrtc, RTC_BKP_DR1);
            uint32_t device_state = HAL_RTCEx_BKUPRead(&hrtc, RTC_BKP_DR2);
            restore_device_state(sensor_config, device_state);
        } else {
            // 首次上电或数据无效，执行完整初始化
            full_system_initialization();
        }
    }
}

**关键状态保存与恢复:**
```c
typedef struct power_context {
    // GPIO状态保存
    uint32_t gpio_moder[9];    // GPIOA~GPIOH
    uint32_t gpio_otyper[9];
    uint32_t gpio_ospeedr[9];
    uint32_t gpio_pupdr[9];
    
    // 时钟配置保存
    uint32_t rcc_cfgr;
    uint32_t rcc_pllcfgr;
    
    // 外设使能状态
    uint32_t rcc_ahb1enr;
    uint32_t rcc_apb1enr;
    uint32_t rcc_apb2enr;
    
    // UART配置保存
    uint32_t uart_cr1;
    uint32_t uart_brr;
    
    // SPI配置保存(LoRa模块)
    uint32_t spi_cr1;
    uint32_t spi_cr2;
} power_context_t;

static power_context_t g_power_context;

void save_peripheral_state(void)
{
    // 保存GPIO状态
    for (int i = 0; i < 9; i++) {
        GPIO_TypeDef *gpio = (GPIO_TypeDef *)(GPIOA_BASE + i * 0x400);
        g_power_context.gpio_moder[i] = gpio->MODER;
        g_power_context.gpio_otyper[i] = gpio->OTYPER;
        g_power_context.gpio_ospeedr[i] = gpio->OSPEEDR;
        g_power_context.gpio_pupdr[i] = gpio->PUPDR;
    }
    
    // 保存时钟配置
    g_power_context.rcc_cfgr = RCC->CFGR;
    g_power_context.rcc_pllcfgr = RCC->PLLCFGR;
    
    // 保存外设使能状态
    g_power_context.rcc_ahb1enr = RCC->AHB1ENR;
    g_power_context.rcc_apb1enr = RCC->APB1ENR;
    g_power_context.rcc_apb2enr = RCC->APB2ENR;
    
    // 保存关键外设配置
    g_power_context.uart_cr1 = USART1->CR1;
    g_power_context.uart_brr = USART1->BRR;
    g_power_context.spi_cr1 = SPI1->CR1;
    g_power_context.spi_cr2 = SPI1->CR2;
}

void restore_peripheral_state(void)
{
    // 恢复时钟配置
    RCC->CFGR = g_power_context.rcc_cfgr;
    RCC->PLLCFGR = g_power_context.rcc_pllcfgr;
    
    // 重新启动PLL
    RCC->CR |= RCC_CR_PLLON;
    while(!(RCC->CR & RCC_CR_PLLRDY));
    
    // 切换到PLL
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL);
    
    // 恢复外设使能
    RCC->AHB1ENR = g_power_context.rcc_ahb1enr;
    RCC->APB1ENR = g_power_context.rcc_apb1enr;
    RCC->APB2ENR = g_power_context.rcc_apb2enr;
    
    // 恢复GPIO配置
    for (int i = 0; i < 9; i++) {
        GPIO_TypeDef *gpio = (GPIO_TypeDef *)(GPIOA_BASE + i * 0x400);
        gpio->MODER = g_power_context.gpio_moder[i];
        gpio->OTYPER = g_power_context.gpio_otyper[i];
        gpio->OSPEEDR = g_power_context.gpio_ospeedr[i];
        gpio->PUPDR = g_power_context.gpio_pupdr[i];
    }
    
    // 恢复关键外设
    USART1->CR1 = g_power_context.uart_cr1;
    USART1->BRR = g_power_context.uart_brr;
    SPI1->CR1 = g_power_context.spi_cr1;
    SPI1->CR2 = g_power_context.spi_cr2;
}
```

```c
    printf("PASS: System state cleaned\r\n");
    
    // 最后的日志
    printf("Jumping to application at 0x%08X...\r\n", app_addr);
    
    // 确保日志输出完成
    while(__HAL_UART_GET_FLAG(&huart1, UART_FLAG_TC) == RESET);
    
    // 执行跳转
    __set_MSP(vector_table[0]);
    void (*app_reset_handler)(void) = (void*)vector_table[1];
    app_reset_handler();
}
```

## 项目背景补充

### BootLoader项目架构
```
Flash布局:
- Sector 0-5: BootLoader程序
- Sector 6: 临时存储区域 (0x08040000)
- Sector 7: 应用程序运行区域 (0x08060000)

工作流程:
1. 接收固件文件到Sector 6
2. 解析mkimage头部信息
3. 将固件重定位到Sector 7
4. 验证并跳转到应用程序
```
